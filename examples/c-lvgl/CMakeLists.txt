cmake_minimum_required(VERSION 3.16)
project(escope_c_lvgl LANGUAGES C)

if(NOT UNIX OR NOT EXISTS "${LVGL_SOURCE}/include/lvgl/lvgl.h")
  message(FATAL_ERROR "A POSIX host and LVGL_SOURCE pointing to LVGL 9.6 source are required")
endif()
if(NOT EXISTS "${ESCOPE_LIBRARY}" OR NOT EXISTS "${ESCOPE_INCLUDE}/escope.h")
  message(FATAL_ERROR "Set ESCOPE_LIBRARY to the host C SDK archive and ESCOPE_INCLUDE to its header directory")
endif()

file(GLOB_RECURSE lvgl_sources CONFIGURE_DEPENDS "${LVGL_SOURCE}/src/*.c")
add_library(lvgl_c STATIC ${lvgl_sources})
target_include_directories(lvgl_c SYSTEM PUBLIC "${LVGL_SOURCE}/include" PRIVATE "${LVGL_SOURCE}")
target_compile_definitions(lvgl_c PUBLIC LV_CONF_SKIP=1)

add_executable(escope-c-lvgl main.c tree.c)
set_target_properties(escope-c-lvgl PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_compile_definitions(escope-c-lvgl PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(escope-c-lvgl PRIVATE -Wall -Wextra -Werror)
target_include_directories(escope-c-lvgl PRIVATE "${ESCOPE_INCLUDE}")
separate_arguments(escope_link_options NATIVE_COMMAND "${ESCOPE_NATIVE_LIBS}")
set(escope_link_items)
set(expect_framework FALSE)
foreach(item IN LISTS escope_link_options)
  if(expect_framework)
    # Keep the name attached to the flag; CMake otherwise treats it as -l<name>.
    list(APPEND escope_link_items "-Wl,-framework,${item}")
    set(expect_framework FALSE)
  elseif(item STREQUAL "-framework")
    set(expect_framework TRUE)
  else()
    list(APPEND escope_link_items "${item}")
  endif()
endforeach()
if(expect_framework)
  message(FATAL_ERROR "Missing framework name in ESCOPE_NATIVE_LIBS")
endif()
target_link_libraries(escope-c-lvgl PRIVATE lvgl_c "${ESCOPE_LIBRARY}" ${escope_link_items} m)
